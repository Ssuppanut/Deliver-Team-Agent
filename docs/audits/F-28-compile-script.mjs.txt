const rows = [];
const emit = (p, a) => {
  const l = (p.fragments?.[a] ?? []).filter((x) => !/Probe[A-Za-z0-9]+/.test(x) && !/^import /.test(x));
  const s = l.join(' ⏎ ');
  return s ? (s.length > 150 ? s.slice(0, 147) + '…' : s) : '(nothing emitted)';
};
const caught = (p, a) => {
  const out = [];
  for (const [g, v] of Object.entries(p.gates ?? {})) {
    if (v.ok) continue;
    const mine = (v.issues ?? []).filter((i) => i.includes(`${a}:`) || !/^\[[a-z]*\] [a-z-]*: (react|vue|svelte|react-native|swiftui|compose):/.test(i));
    if (g === 'declared-io') continue; // probe artifact: unused declared props in my minimal specs
    if (mine.length) out.push(g);
  }
  return out.join(',') || '';
};
function row(p, a, o) {
  rows.push({ id: p.id, construct: p.construct, value: p.value, adapter: a, emitted: emit(p, a), expected: o.exp, verdict: o.v, severity: o.v === 'CORRECT' ? '-' : o.sev, conf: o.conf ?? 'E', refs: o.refs ?? '', note: o.note ?? '', caught: caught(p, a) });
}
const web = (p, o) => WEB.forEach((a) => row(p, a, o));
